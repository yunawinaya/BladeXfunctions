// ==========================================================================
// ITEM BUNDLES
// --------------------------------------------------------------------------
// An item bundle is ONE line on the order with the bundle's items beneath it,
// and it is invoiced as a whole. The order hands that tree over in one of two
// shapes -- the items nested under the bundle line's `children`, or flat rows
// of table_so pointing back through parent_id -- so both are read here and
// turned into one list with each bundle's items directly after their bundle.
//
// table_si is then built with the same shape: the bundle is one row carrying
// its items under `children`, which is exactly what the form hands SI_SAVE
// when a bundle is added there. Without it the items land beside the bundle as
// loose lines, the invoice shows no bundle at all, and SI_SAVE has nothing to
// hang them under.
//
// A bundle line carries no item of its own, so it has no material, no item
// category, no UOM and no tax rate. Those are id-typed columns on the invoice
// and a null reaching one comes back as a BigInt conversion failure, so they
// are blanked rather than stringified into "null" / "undefined". so_discount is
// never written on a bundle line either, and parseFloat(undefined) would put
// the string "NaN" into a numeric column.
// ==========================================================================
const referenceId = (value) => {
  if (value && typeof value === "object") return value.id || null;
  return value || null;
};

// The order's lines as one list: each bundle's items directly after their
// bundle line, linked back to it by parentId. A row that sits under a bundle is
// not repeated as a line of its own.
const flattenSoLines = (lines) => {
  const rows = lines || [];

  const childrenOf = (line) => {
    const nested = Array.isArray(line.children) ? line.children : [];
    if (nested.length > 0) return nested;

    return rows.filter(
      (row) =>
        row.parent_id !== undefined &&
        row.parent_id !== null &&
        row.parent_id !== "" &&
        String(row.parent_id) === String(line.id),
    );
  };

  const claimed = new Set();

  rows.forEach((line) => {
    childrenOf(line).forEach((child) => {
      if (child && child.id !== undefined && child.id !== null) {
        claimed.add(String(child.id));
      }
    });
  });

  const flat = [];

  rows.forEach((line) => {
    if (claimed.has(String(line.id))) return;

    flat.push({ line: line, parentId: null });

    childrenOf(line).forEach((child) => {
      flat.push({ line: child, parentId: String(line.id) });
    });
  });

  return flat;
};

// The invoice date and time on the engine's own clock, "YYYY-MM-DD HH:mm:ss".
// toISOString() would shift it to UTC -- an hour that reads wrong and, before
// 08:00 local, a calendar day early -- and toLocaleString() hands back
// "Wed Aug 26 2026 10:41:38 GMT+0800 (...)", which Add Data cannot parse into
// a TimeStamp at all. The local getters read the same clock the user sees.
const localDateTimeString = (date) => {
  const pad = (n) => String(n).padStart(2, "0");

  const day = [
    date.getFullYear(),
    pad(date.getMonth() + 1),
    pad(date.getDate()),
  ].join("-");

  const time = [
    pad(date.getHours()),
    pad(date.getMinutes()),
    pad(date.getSeconds()),
  ].join(":");

  return `${day} ${time}`;
};

// Helper function to calculate totals from SO data
