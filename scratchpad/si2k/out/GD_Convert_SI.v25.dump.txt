// Helper function to calculate totals from GD lines
const calculateTotalsFromGdLines = (gdLines) => {
  let totals = {
    invoice_subtotal: 0,
    invoice_total_discount: 0,
    invoice_taxes_amount: 0,
    invoice_total: 0,
    myr_total_amount: 0,
  };

  for (const gdLine of gdLines) {
    totals.invoice_subtotal += parseFloat(gdLine.recalculated_gross) || 0;
    totals.invoice_total_discount +=
      parseFloat(gdLine.recalculated_discount_amount) || 0;
    totals.invoice_taxes_amount +=
      parseFloat(gdLine.recalculated_tax_amount) || 0;
    totals.invoice_total += parseFloat(gdLine.recalculated_amount) || 0;
    totals.myr_total_amount += parseFloat(gdLine.myr_amount) || 0;
  }

  totals.invoice_subtotal = totals.invoice_subtotal.toFixed(8);
  totals.invoice_total_discount = totals.invoice_total_discount.toFixed(8);
  totals.invoice_taxes_amount = totals.invoice_taxes_amount.toFixed(8);
  totals.invoice_total = totals.invoice_total.toFixed(8);
  totals.myr_total_amount = totals.myr_total_amount.toFixed(8);

  return totals;
};

// Process GD list and create SI
const processGdList = (gdListData) => {
  const siLineItems = [];
  let lineCounter = 0;
  let allGdLines = [];
  const plantID = gdListData[0].plant_id;

  for (const gd of gdListData) {
    // Get GD lines
    const gdLines = gd.table_gd || [];

    // Every line billed so far that an item could hang off, by the delivery
    // line it came from. flattenGdLines has already put each bundle's items
    // directly after their bundle, so the row an item points at has always been
    // reached first.
    const billedByGdLine = {};

    for (const entry of flattenGdLines(gdLines)) {
      const gdItem = entry.line;
      const bundle = entry.parentId ? billedByGdLine[entry.parentId] : null;

      // An item under a bundle that was not billed is not billed either: a
      // bundle is invoiced whole or not at all.
      if (entry.parentId && !bundle) continue;

      // Find matching SO line item
      const soLineItem = findSoLineItem(gdItem, soLineItemList);
      const soData = findSOData(gdItem, soList);

      // A GD line with no SO line behind it (description row) carries no price -- nothing to bill.
      if (!soLineItem || !soData) continue;

      // A cash order's line is already billed by the order itself.
      if (soData.so_type === "Cash") continue;

      const soQuantity = parseFloat(soLineItem.so_quantity) || 0;
      const alreadyInvoiced = parseFloat(soLineItem.invoice_qty) || 0;
      // Returned goods are not billable under the deduct method, so they come
      // off the order line's remaining allowance too.
      const soReturned = deductReturnQty
        ? parseFloat(soLineItem.return_qty) || 0
        : 0;
      // What this SO line still allows to be invoiced. SI_SAVE rejects (410) anything above it.
      const remaining = Math.max(0, q8(soQuantity - alreadyInvoiced - soReturned));

      let invoiceQty = parseFloat(gdItem.gd_qty) || 0;

      // A bundle's items move with their bundle, so only a line measured on its
      // own -- a top-level line, or the bundle row itself -- has its return taken
      // off here; the items then scale by the bundle's ratio as usual.
      if (deductReturnQty && !bundle) {
        invoiceQty = Math.max(
          0,
          q8(invoiceQty - (parseFloat(gdItem.return_qty) || 0)),
        );
        // Nothing left to bill, and a zero-quantity line is what SI_SAVE rejects with 407.
        if (invoiceQty <= 0) continue;
      }

      if (bundle) {
        // Invoicing part of a bundle invoices the same part of every item in
        // it. The bundle line above already decided how much, so an item is
        // never measured against its own remaining -- it moves with the bundle,
        // which is what keeps the two in step on the order.
        invoiceQty = q8(invoiceQty * bundle.ratio);
      } else if (isAuto) {
        // Already fully invoiced by an earlier GD on the same SO line -- skip, do not error.
        if (remaining <= 0) continue;
        // Over-delivery beyond the ordered quantity cannot be invoiced against this SO line.
        if (invoiceQty > remaining) invoiceQty = remaining;
        if (invoiceQty <= 0) continue;
      }

      const unitPrice = parseFloat(soLineItem.so_item_price) || 0;
      const discountAmount = parseFloat(soLineItem.so_discount_amount) || 0;
      const taxAmount = parseFloat(soLineItem.so_tax_amount) || 0;
      const totalAmount = parseFloat(soLineItem.so_amount) || 0;

      const ratio = soQuantity > 0 ? invoiceQty / soQuantity : 0;
      const recalculatedGross = unitPrice * invoiceQty;
      const recalculatedDiscountAmount = discountAmount * ratio;
      const recalculatedTaxAmount = taxAmount * ratio;
      const recalculatedAmount = totalAmount * ratio;

      // A bundle's items are numbered inside their bundle -- 1, 1-1, 1-2, 2 --
      // which is the convention SI_SAVE's fillback uses.
      let lineIndex;

      if (bundle) {
        bundle.itemsPlaced++;
        lineIndex = `${bundle.lineIndex}-${bundle.itemsPlaced}`;
      } else {
        lineCounter++;
        lineIndex = lineCounter;
      }

      const lineItem = mapLineItem(
        gdItem,
        soLineItem,
        gd,
        lineIndex,
        soData,
        plantID,
        invoiceQty,
      );

      if (bundle) {
        // The item is a row of its bundle, not a row of the invoice.
        if (!Array.isArray(bundle.row.children)) bundle.row.children = [];

        bundle.row.children.push(lineItem);
      } else {
        const deliveredBundles = parseFloat(gdItem.gd_qty) || 0;

        billedByGdLine[String(gdItem.id)] = {
          row: lineItem,
          lineIndex,
          itemsPlaced: 0,
          // How much of the bundle is being invoiced. 1 on an ordinary convert;
          // less only where the auto path capped the bundle line above.
          ratio: deliveredBundles > 0 ? invoiceQty / deliveredBundles : 0,
        };

        // Store recalculated values for totals calculation. Only a top-level
        // line is counted: a bundle line's gross, discount, tax and amount on
        // the order are already the sum of its items, so counting the items as
        // well bills the bundle twice.
        gdItem.recalculated_gross = recalculatedGross;
        gdItem.recalculated_discount_amount = recalculatedDiscountAmount;
        gdItem.recalculated_tax_amount = recalculatedTaxAmount;
        gdItem.recalculated_amount = recalculatedAmount;
        gdItem.myr_amount =
          recalculatedAmount * (parseFloat(soData.exchange_rate) || 1);

        allGdLines.push(gdItem);
        siLineItems.push(lineItem);
      }
    }
  }

  return { siLineItems, allGdLines };
};

    (created.length === 1 ? "" : "s") +
