// A bundle line has no item of its own; the items under it carry both the
// bundle and an item. table_si reaches this node in either of two shapes --
// nested under `children`, or flat with a bundle's items following their bundle
// line -- so both are read here. Each entry remembers the position of its
// bundle line (-1 when the row is not in a bundle) and whether it was nested to
// begin with, so what is written back keeps the shape it arrived in.
const bundleKey = (row) => {
  const bundle = row.item_bundle_id;

  if (bundle && typeof bundle === "object") {
    return String(bundle.id || bundle.ib_no || "");
  }

  return String(bundle || "");
};

const flattenWithParent = (rows, itemKey) => {
  const flat = [];
  const positionOf = new Map();

  const isBundleRow = (row) => Boolean(bundleKey(row)) && !row[itemKey];
  const isBundleItem = (row) => Boolean(bundleKey(row)) && Boolean(row[itemKey]);

  let openBundle = -1; // bundle line the flat rows below it belong to

  for (const [index, row] of (rows || []).entries()) {
    const position = flat.length;
    positionOf.set(index, position);

    const children = Array.isArray(row.children) ? row.children : [];

    // Already a tree: nothing to reattach, and a bundle spelled out this way
    // ends whatever run was open above it.
    if (children.length > 0) {
      openBundle = -1;
      flat.push({ row, parent: -1, nested: false });

      for (const child of children) {
        flat.push({ row: child, parent: position, nested: true });
      }

      continue;
    }

    if (isBundleRow(row)) {
      openBundle = position;
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    if (!isBundleItem(row)) {
      openBundle = -1; // a plain line closes the bundle above it
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    // A flat item: linked to its bundle line by id, or else belonging to the
    // bundle line it follows, as long as that is the same bundle.
    let parent = -1;

    for (let candidate = index - 1; candidate >= 0; candidate--) {
      const candidateRow = rows[candidate];

      if (!isBundleRow(candidateRow)) continue;

      const byId =
        row.parent_id != null &&
        candidateRow.id != null &&
        String(row.parent_id) === String(candidateRow.id);
      const byKey =
        row.parent_fm_key != null &&
        candidateRow.fm_key != null &&
        String(row.parent_fm_key) === String(candidateRow.fm_key);

      if (byId || byKey) {
        parent = positionOf.get(candidate);
        break;
      }
    }

    if (
      parent === -1 &&
      openBundle !== -1 &&
      bundleKey(flat[openBundle].row) === bundleKey(row)
    ) {
      parent = openBundle;
    }

    flat.push({ row, parent, nested: false });
  }

  return flat;
};

      parent = openBundle;
    }

    flat.push({ row, parent, nested: false });
  }

  return flat;
};

const flattenRows = (rows, itemKey) =>
  flattenWithParent(rows, itemKey).map((item) => item.row);


const validateInvoiceQuantities = (items, tableSi, docId, docIdKey, siLineIdKey, qtyKey, itemKey) => {
  // A bundle's items are invoiced lines in their own right -- each has its own
  // order line and its own invoiced quantity -- so both sides are read flat.
  const flatItems = flattenRows(items, itemKey);
  const filteredSI = flattenRows(tableSi, "material_id").filter((item) => item[docIdKey] === docId);
  
  const errors = [];
  
  filteredSI.forEach((siItem) => {
    const originalItem = flatItems.find(item => item.id === siItem[siLineIdKey]);
    
    if (originalItem) {
      // Clamped: a line can be returned in full.
      const returnedQty = deductReturnQty
        ? absRound8(originalItem.return_qty)
        : 0;
      const availableQty = Math.max(
        0,
        absRound8(originalItem[qtyKey]) - returnedQty,
      );
      const currentInvoicedQty = absRound8(originalItem.invoice_qty);
      const newInvoiceQty = absRound8(siItem.invoice_qty);
      const totalInvoicedQty = absRound8(currentInvoicedQty + newInvoiceQty);
      
      if (totalInvoicedQty > availableQty) {
        errors.push({
          itemName: originalItem.item_id || originalItem.material_name || `Item ${originalItem.id}`,
          // Carried so the message can say what the cap is and where it came
          // from: the same line is checked against both the delivery and the
          // order, and "10 delivered" and "10 ordered" are different sentences.
          available: availableQty,
          total: absRound8(originalItem[qtyKey]),
          basis: qtyKey === "gd_qty" ? "delivered" : "ordered",
        });
      }
    }
  });
  
  return errors;
};

const updateGoodsDelivery = (gdIds, tableSi) => {
  let gdUpdatedList = [];
  for(const gdID of gdIds) {
    const gdData = gdList.find(gd => gd.id === gdID)
    const gdItems = gdData.table_gd || [];

      "gd_line_id",
      "gd_qty",

const processItems = (items, tableSi, docId, docIdKey, siLineIdKey, qtyKey, itemKey) => {
  // Filter PI items for the current document -- a bundle's items are invoiced
  // lines of their own, so the invoice is read flat here.
  const filteredSI = flattenRows(tableSi, "material_id").filter((item) => item[docIdKey] === docId);

  // The order's own lines, flat, each remembering the bundle line it sits under.
  const flat = flattenWithParent(items, itemKey);

  // Filter items where item.id matches any siLineIdKey in filteredSI
  const filteredItems = flat.filter((entry) =>
    filteredSI.some((pi) => pi[siLineIdKey] === entry.row.id),
  );

  // Initialize tracking. Only a bundle line counts towards the document's
  // status -- its items cannot be invoiced independently of it.
  let totalItems = flat.filter((entry) => entry.parent === -1).length;
  let partiallyInvoicedItems = 0;
  let fullyInvoicedItems = 0;
  const updated = flat.map((entry) => ({ ...entry, row: { ...entry.row } }));

  // Update invoice quantities
  filteredItems.forEach((filteredItem, filteredIndex) => {
    const originalIndex = updated.findIndex(
      (entry) => entry.row.id === filteredItem.row.id,
    );

    if (originalIndex === -1) return;

    const itemQty = parseFloat(filteredItem.row[qtyKey] || 0);
    const piInvoicedQty = absRound8(
      filteredSI[filteredIndex]?.invoice_qty
    );
    const currentInvoicedQty = absRound8(
      updated[originalIndex].row.invoice_qty
    );
    const totalInvoicedQty = absRound8(currentInvoicedQty + piInvoicedQty);

    // Update invoice quantity
    updated[originalIndex].row.invoice_qty = totalInvoicedQty;
  });

  for (const entry of updated) {
    if (entry.parent !== -1) continue;

    // Under the deduct method a returned quantity is settled by never billing it.
    const billable = Math.max(
      0,
      absRound8(entry.row[qtyKey]) -
        (deductReturnQty ? absRound8(entry.row.return_qty) : 0),
    );

    if (absRound8(entry.row.invoice_qty) > 0) {
      partiallyInvoicedItems++;
      entry.row.si_status = "Partially Invoiced";
      if (absRound8(entry.row.invoice_qty) >= billable) {
        fullyInvoicedItems++;
        entry.row.si_status = "Fully Invoiced";
      }
    } else if (deductReturnQty && billable <= 0) {
      fullyInvoicedItems++;
    }
  }

  // A bundle's items carry the bundle's status, whatever it came out as.
  for (const entry of updated) {
    if (entry.parent === -1) continue;

    entry.row.si_status = updated[entry.parent].row.si_status;
  }

  // Written back in the shape it arrived in: a bundle's items go back under
  // their bundle line only if that is where they came from.
  const updatedItems = [];

  for (const [index, entry] of updated.entries()) {
    if (entry.nested) continue;

    const children = updated
      .filter((candidate) => candidate.nested && candidate.parent === index)
      .map((candidate) => candidate.row);

    if (children.length > 0) {
      updatedItems.push({ ...entry.row, children });
    } else if (Array.isArray(entry.row.children)) {
      updatedItems.push({ ...entry.row, children: [] });
    } else {
      updatedItems.push(entry.row);
    }
  }

  // Determine new PI status
  const allItemsComplete = fullyInvoicedItems === totalItems;
  const anyItemProcessing = partiallyInvoicedItems > 0;
  let newSIStatus = anyItemProcessing
    ? allItemsComplete
      ? "Fully Invoiced"
      : "Partially Invoiced"
    : "";

  return { updatedItems, newSIStatus };
};

// A bundle line has no item of its own; the items under it carry both the
// bundle and an item. table_si, table_so and table_gd reaches this node in either of two shapes --
// nested under `children`, or flat with a bundle's items following their bundle
// line -- so both are read here. Each entry remembers the position of its
// bundle line (-1 when the row is not in a bundle) and whether it was nested to
// begin with, so what is written back keeps the shape it arrived in.
const bundleKey = (row) => {
  const bundle = row.item_bundle_id;

  if (bundle && typeof bundle === "object") {
    return String(bundle.id || bundle.ib_no || "");
  }

  return String(bundle || "");
};

const flattenWithParent = (rows, itemKey) => {
  const flat = [];
  const positionOf = new Map();

  const isBundleRow = (row) => Boolean(bundleKey(row)) && !row[itemKey];
  const isBundleItem = (row) => Boolean(bundleKey(row)) && Boolean(row[itemKey]);

  let openBundle = -1; // bundle line the flat rows below it belong to

  for (const [index, row] of (rows || []).entries()) {
    const position = flat.length;
    positionOf.set(index, position);

    const children = Array.isArray(row.children) ? row.children : [];

    // Already a tree: nothing to reattach, and a bundle spelled out this way
    // ends whatever run was open above it.
    if (children.length > 0) {
      openBundle = -1;
      flat.push({ row, parent: -1, nested: false });

      for (const child of children) {
        flat.push({ row: child, parent: position, nested: true });
      }

      continue;
    }

    if (isBundleRow(row)) {
      openBundle = position;
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    if (!isBundleItem(row)) {
      openBundle = -1; // a plain line closes the bundle above it
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    // A flat item: linked to its bundle line by id, or else belonging to the
    // bundle line it follows, as long as that is the same bundle.
    let parent = -1;

    for (let candidate = index - 1; candidate >= 0; candidate--) {
      const candidateRow = rows[candidate];

      if (!isBundleRow(candidateRow)) continue;

      const byId =
        row.parent_id != null &&
        candidateRow.id != null &&
        String(row.parent_id) === String(candidateRow.id);
      const byKey =
        row.parent_fm_key != null &&
        candidateRow.fm_key != null &&
        String(row.parent_fm_key) === String(candidateRow.fm_key);

      if (byId || byKey) {
        parent = positionOf.get(candidate);
        break;
      }
    }

    if (
      parent === -1 &&
      openBundle !== -1 &&
      bundleKey(flat[openBundle].row) === bundleKey(row)
    ) {
      parent = openBundle;
    }

    flat.push({ row, parent, nested: false });
  }

  return flat;
};

    : "";

  return { updatedItems, newSIStatus };
};

const flattenWithParent = (rows, itemKey) => {
  const flat = [];
  const positionOf = new Map();

  const isBundleRow = (row) => Boolean(bundleKey(row)) && !row[itemKey];
  const isBundleItem = (row) => Boolean(bundleKey(row)) && Boolean(row[itemKey]);

  let openBundle = -1; // bundle line the flat rows below it belong to

  for (const [index, row] of (rows || []).entries()) {
    const position = flat.length;
    positionOf.set(index, position);

    const children = Array.isArray(row.children) ? row.children : [];

    // Already a tree: nothing to reattach, and a bundle spelled out this way
    // ends whatever run was open above it.
    if (children.length > 0) {
      openBundle = -1;
      flat.push({ row, parent: -1, nested: false });

      for (const child of children) {
        flat.push({ row: child, parent: position, nested: true });
      }

      continue;
    }

    if (isBundleRow(row)) {
      openBundle = position;
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    if (!isBundleItem(row)) {
      openBundle = -1; // a plain line closes the bundle above it
      flat.push({ row, parent: -1, nested: false });
      continue;
    }

    // A flat item: linked to its bundle line by id, or else belonging to the
    // bundle line it follows, as long as that is the same bundle.
    let parent = -1;

    for (let candidate = index - 1; candidate >= 0; candidate--) {
      const candidateRow = rows[candidate];

      if (!isBundleRow(candidateRow)) continue;

        row.parent_id != null &&
        candidateRow.id != null &&
        String(row.parent_id) === String(candidateRow.id);
      const byKey =
        row.parent_fm_key != null &&
        candidateRow.fm_key != null &&
        String(row.parent_fm_key) === String(candidateRow.fm_key);

      if (byId || byKey) {
        parent = positionOf.get(candidate);
        break;
      }
    }

    if (
      parent === -1 &&
      openBundle !== -1 &&
      bundleKey(flat[openBundle].row) === bundleKey(row)

      parent = openBundle;
    }

    flat.push({ row, parent, nested: false });
